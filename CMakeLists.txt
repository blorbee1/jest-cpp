cmake_minimum_required(VERSION 4.2)
project(jest_cpp VERSION 1.0.0 DESCRIPTION "C++ port of JestJS" LANGUAGES CXX)

add_library(jest-cpp INTERFACE)
add_library(JestCpp::JestCpp ALIAS jest-cpp)

target_compile_features(jest-cpp INTERFACE cxx_std_20)
target_include_directories(jest-cpp INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_options(jest-cpp INTERFACE $<$<CXX_COMPILER_ID:MSVC>:/Zc:preprocessor>)

# separate compilation (default)
option(JEST_CPP_SEPARATE_COMPILATION "Compile the std::regex parts once, not in every test file" ON)
if (JEST_CPP_SEPARATE_COMPILATION)
    add_library(jest-cpp-impl STATIC src/jest-cpp.cpp)
    target_include_directories(jest-cpp-impl PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
    target_compile_features(jest-cpp-impl PUBLIC cxx_std_20)
    target_compile_definitions(jest-cpp-impl PUBLIC JEST_CPP_SEPARATE)
    target_link_libraries(jest-cpp INTERFACE jest-cpp-impl)
endif ()

install(FILES include/jest-cpp.hpp DESTINATION include)

# other builds
if (CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    option(JEST_CPP_BUILD_TESTS "Build tests" ON)
    option(JEST_CPP_BUILD_EXAMPLES "Build examples" ON)
    option(JEST_CPP_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)

    function(jest_cpp_enable_warnings target)
        if (MSVC)
            target_compile_options(${target} PRIVATE /W4 /permissive-)
            if (JEST_CPP_WARNINGS_AS_ERRORS)
                target_compile_options(${target} PRIVATE /WX)
            endif ()
        else ()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Wconversion)
            if (JEST_CPP_WARNINGS_AS_ERRORS)
                target_compile_options(${target} PRIVATE -Werror)
            endif ()
        endif ()
    endfunction()

    if (JEST_CPP_BUILD_TESTS)
        enable_testing()
        add_subdirectory(tests)
    endif ()

    if (JEST_CPP_BUILD_EXAMPLES)
        add_subdirectory(examples)
    endif ()
endif ()
